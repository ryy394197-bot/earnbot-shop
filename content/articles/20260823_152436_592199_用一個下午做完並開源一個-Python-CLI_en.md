```python
# 建立一個簡單的命令列介面
import click

@click.command()
def hello():
    """簡單的 Hello World 命令"""
    print("Hello, world!")

if __name__ == "__main__":
    hello()

# 調整程式碼並優化功能
from datetime import datetime

@click.command()
@click.option('--date', default=datetime.now().strftime("%Y-%m-%d"), help='輸入日期格式')
def show_date(date):
    """顯示指定的日期"""
    print(f"今天是 {date}")

# 使用 Python 的函式庫實現更複雜的功能
import requests

@click.command()
@click.argument('url')
def fetch(url):
    """從網路上抓取資料"""
    response = requests.get(url)
    if response.status_code == 200:
        print(response.text)

# 探討如何使用 Markdown 來撰寫使用指南與進階技巧包
import click

@click.command()
@click.argument('filename')
def write_guide(filename):
    """將程式碼轉換為 Markdown 使用指南"""
    with open(filename, 'w') as f:
        f.write('# Python CLI\n\n## Hello World\n\n```python\ndef hello():\n    print("Hello, world!")\n```\n')

# 如何在 GitHub 上公開你的專案
import os

def push_to_github(repo_name):
    """將專案推送到 GitHub"""
    os.system(f'git init')
    os.system(f'git add .')
    os.system(f'git commit -m "Initial commit"')
    os.system(f'git remote add origin https://github.com/username/{repo_name}.git')
    os.system('git push -u origin master')
```
