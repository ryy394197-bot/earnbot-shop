```python
def read_csv(file_path):
    with open(file_path, 'r') as file:
        reader = csv.DictReader(file)
        data = [OrderedDict(row) for row in reader]
```

```python
def convert_to_json(data):
    json_data = json.dumps(data, indent=4, sort_keys=True)
    return json_data
```

```python
def main():
```
