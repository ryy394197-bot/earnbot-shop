#### 1. 實作 CLI 工具
```python
import os
from datetime import datetime

def organize_photos(source_dir, target_dir):
    # 檢查目標資料夾是否存在，若不存在則建立
    if not os.path.exists(target_dir):
        os.makedirs(target_dir)

    for filename in os.listdir(source_dir):
        # 只處理圖片檔案
        if not filename.lower().endswith(('.jpg', '.jpeg', '.png')):
            continue

        photo_path = os.path.join(source_dir, filename)
        timestamp = datetime.fromtimestamp(os.path.getmtime(photo_path))
        target_date = timestamp.strftime('%Y-%m-%d')

        destination = os.path.join(target_dir, target_date)
        if not os.path.exists(destination):
            os.makedirs(destination)

        final_destination = os.path.join(destination, filename)
        os.rename(photo_path, final_destination)

if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Organize photos by date")
    parser.add_argument('source_dir', type=str, help='Source directory containing photos')
    parser.add_argument('--target-dir', '-t', default='./organized', 
                        type=str, help='Target directory to store organized photos (default: ./organized)')

    args = parser.parse_args()

    organize_photos(args.source_dir, args.target_dir)
```
