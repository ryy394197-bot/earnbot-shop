```python
import os
from datetime import datetime

class Todo:
    def __init__(self):
        self.todo_list = []

    def add_todo(self, todo):
        now = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        self.todo_list.append({'todo': todo, 'timestamp': now})
        print(f'已添加 "{todo}" 至清單，時間戳記：{now}')

    def remove_todo(self, index):
        if 0 <= index < len(self.todo_list):
            removed_todo = self.todo_list.pop(index)
            print(f'已刪除索引 {index} 的待辦事項："{removed_todo["todo"]}"')
        else:
            print('索引超出範圍')

    def view_todos(self):
        if not self.todo_list:
            print('清單為空')
            return
        for index, todo in enumerate(self.todo_list):
            timestamp = todo['timestamp']
            print(f'{index + 1}. {todo["todo"]} - {timestamp}')
```
